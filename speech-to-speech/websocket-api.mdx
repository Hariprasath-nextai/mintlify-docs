## Quick Start

Connect directly to NextEVI's WebSocket API for maximum control and platform flexibility.

### Connection

```javascript
const ws = new WebSocket(
  `wss://api.nextevi.com/voice/websocket/${connectionId}?api_key=oak_your_api_key&config_id=your_config_id&project_id=your_project_id`
);

ws.onopen = () => {
  console.log('Connected to NextEVI');
  
  // Configure session
  ws.send(JSON.stringify({
    type: "session_settings",
    timestamp: Date.now() / 1000,
    message_id: "settings-1",
    data: {
      emotion_detection: { enabled: true },
      turn_detection: { enabled: true, silence_threshold: 0.5 },
      audio: { sample_rate: 24000, channels: 1, encoding: "linear16" }
    }
  }));
};
```

## Core Message Types

### Send Audio (Client → Server)

```javascript
// Send PCM audio data
ws.send(JSON.stringify({
  type: "audio_input",
  timestamp: Date.now() / 1000,
  message_id: `audio-${Date.now()}`,
  data: {
    audio: btoa(audioData),  // Base64-encoded PCM
    chunk_id: chunkNumber,
    sample_rate: 24000,
    channels: 1,
    encoding: "linear16"
  }
}));
```

### Receive Messages (Server → Client)

```javascript
ws.onmessage = (event) => {
  const message = JSON.parse(event.data);
  
  switch (message.type) {
    case 'transcription':
      // User speech transcribed
      console.log('User said:', message.data.transcript);
      console.log('Emotions:', message.data.emotions);
      break;
      
    case 'llm_response_chunk':
      // AI response text (streaming)
      console.log('AI response:', message.data.content);
      break;
      
    case 'audio_output':
      // TTS audio data
      playAudio(message.data.audio); // Base64 audio
      break;
      
    case 'emotion_update':
      // Real-time emotion detection
      console.log('Dominant emotion:', message.data.dominant_emotion);
      console.log('Confidence:', message.data.confidence);
      break;
      
    case 'status':
      // System status updates
      handleStatusUpdate(message.data.status);
      break;
      
    case 'error':
      // Error handling
      console.error('Error:', message.data.error_message);
      break;
  }
};
```

## Complete Working Example

Here's a full WebSocket voice chat implementation:

```javascript
class NextEVIWebSocket {
  constructor(config) {
    this.config = config;
    this.ws = null;
    this.isConnected = false;
    this.audioContext = null;
    this.mediaRecorder = null;
  }

  async connect() {
    const url = `wss://api.nextevi.com/voice/websocket/${this.config.connectionId}?api_key=${this.config.apiKey}&config_id=${this.config.configId}&project_id=${this.config.projectId}`;
    
    this.ws = new WebSocket(url);
    
    this.ws.onopen = () => {
      console.log('Connected to NextEVI');
      this.isConnected = true;
      
      // Configure session
      this.send({
        type: "session_settings",
        data: {
          emotion_detection: { enabled: true, confidence_threshold: 0.7 },
          turn_detection: { enabled: true, silence_threshold: 0.5 },
          audio: { sample_rate: 24000, channels: 1, encoding: "linear16" }
        }
      });
      
      // Start audio capture
      this.startAudioCapture();
    };
    
    this.ws.onmessage = (event) => {
      const message = JSON.parse(event.data);
      this.handleMessage(message);
    };
    
    this.ws.onclose = () => {
      console.log('Disconnected from NextEVI');
      this.isConnected = false;
      this.stopAudioCapture();
    };
    
    this.ws.onerror = (error) => {
      console.error('WebSocket error:', error);
    };
  }

  send(message) {
    if (this.ws && this.isConnected) {
      this.ws.send(JSON.stringify({
        ...message,
        timestamp: Date.now() / 1000,
        message_id: `msg-${Date.now()}-${Math.random()}`
      }));
    }
  }

  handleMessage(message) {
    switch (message.type) {
      case 'transcription':
        this.onTranscription?.(message.data);
        break;
      case 'llm_response_chunk':
        this.onLLMResponse?.(message.data);
        break;
      case 'audio_output':
        this.playAudio(message.data.audio);
        break;
      case 'emotion_update':
        this.onEmotionUpdate?.(message.data);
        break;
      case 'status':
        this.onStatusUpdate?.(message.data);
        break;
      case 'error':
        this.onError?.(message.data);
        break;
    }
  }

  async startAudioCapture() {
    try {
      const stream = await navigator.mediaDevices.getUserMedia({ 
        audio: {
          sampleRate: 24000,
          channelCount: 1,
          echoCancellation: true,
          noiseSuppression: true
        } 
      });
      
      this.audioContext = new (window.AudioContext || window.webkitAudioContext)({
        sampleRate: 24000
      });
      
      const source = this.audioContext.createMediaStreamSource(stream);
      const processor = this.audioContext.createScriptProcessor(1024, 1, 1);
      
      processor.onaudioprocess = (event) => {
        if (this.isConnected) {
          const audioData = event.inputBuffer.getChannelData(0);
          const int16Data = this.floatTo16BitPCM(audioData);
          
          this.send({
            type: "audio_input",
            data: {
              audio: btoa(String.fromCharCode.apply(null, int16Data)),
              sample_rate: 24000,
              channels: 1,
              encoding: "linear16"
            }
          });
        }
      };
      
      source.connect(processor);
      processor.connect(this.audioContext.destination);
      
    } catch (error) {
      console.error('Failed to start audio capture:', error);
      this.onError?.({ error_message: 'Microphone access denied' });
    }
  }

  floatTo16BitPCM(input) {
    const buffer = new ArrayBuffer(input.length * 2);
    const output = new DataView(buffer);
    for (let i = 0; i < input.length; i++) {
      const s = Math.max(-1, Math.min(1, input[i]));
      output.setInt16(i * 2, s < 0 ? s * 0x8000 : s * 0x7FFF, true);
    }
    return new Uint8Array(buffer);
  }

  async playAudio(base64Audio) {
    try {
      const audioData = atob(base64Audio);
      const arrayBuffer = new ArrayBuffer(audioData.length);
      const view = new Uint8Array(arrayBuffer);
      
      for (let i = 0; i < audioData.length; i++) {
        view[i] = audioData.charCodeAt(i);
      }
      
      const audioBuffer = await this.audioContext.decodeAudioData(arrayBuffer);
      const source = this.audioContext.createBufferSource();
      source.buffer = audioBuffer;
      source.connect(this.audioContext.destination);
      source.start();
      
    } catch (error) {
      console.error('Failed to play audio:', error);
    }
  }

  stopAudioCapture() {
    if (this.audioContext) {
      this.audioContext.close();
      this.audioContext = null;
    }
  }

  disconnect() {
    if (this.ws) {
      this.ws.close();
    }
    this.stopAudioCapture();
  }
}

// Usage
const voice = new NextEVIWebSocket({
  connectionId: 'conn-123',
  apiKey: 'oak_your_api_key',
  configId: 'your-config-id',
  projectId: 'your-project-id'
});

// Set up event handlers
voice.onTranscription = (data) => {
  console.log('User said:', data.transcript);
  if (data.emotions) {
    console.log('Emotions:', data.emotions);
  }
};

voice.onLLMResponse = (data) => {
  console.log('AI response:', data.content);
};

voice.onEmotionUpdate = (data) => {
  console.log('Emotion detected:', data.dominant_emotion, data.confidence);
};

voice.onError = (error) => {
  console.error('Voice error:', error.error_message);
};

// Connect
voice.connect();
```

## Session Configuration

Configure advanced features:

```javascript
ws.send(JSON.stringify({
  type: "session_settings",
  timestamp: Date.now() / 1000,
  message_id: "settings-1",
  data: {
    // Audio settings
    audio: {
      sample_rate: 24000,
      channels: 1,
      encoding: "linear16"
    },
    
    // Turn detection
    turn_detection: {
      enabled: true,
      silence_threshold: 0.5,      // Seconds of silence to end turn
      min_speaking_time: 0.3,      // Minimum speech duration
      interrupt_threshold: 0.8,    // Interrupt sensitivity
      natural_pauses: true         // Allow natural conversation pauses
    },
    
    // Emotion detection
    emotion_detection: {
      enabled: true,
      confidence_threshold: 0.7,   // Minimum confidence to report
      update_frequency: "continuous"
    },
    
    // Idle timeout
    idle_timeout: {
      enabled: true,
      timeout_seconds: 300,        // 5 minutes
      warning_seconds: 30          // 30 second warning
    }
  }
}));
```

## Error Handling

Common error codes and handling:

```javascript
ws.onmessage = (event) => {
  const message = JSON.parse(event.data);
  
  if (message.type === 'error') {
    const { error_code, error_message, recoverable, retry_after } = message.data;
    
    switch (error_code) {
      case 'AUTH_INVALID':
        console.error('Authentication failed - check API key');
        break;
      case 'RATE_LIMITED':
        console.warn(`Rate limited - retry after ${retry_after}ms`);
        setTimeout(() => retryConnection(), retry_after);
        break;
      case 'TRANSCRIPTION_FAILED':
        console.warn('Transcription failed - check audio quality');
        break;
      case 'MICROPHONE_ERROR':
        console.error('Microphone access required');
        requestMicrophonePermission();
        break;
      default:
        console.error('Voice error:', error_message);
    }
  }
};
```

## Best Practices

<AccordionGroup>
  <Accordion title="Connection Management" icon="plug">
    - Implement exponential backoff for reconnection attempts
    - Handle network interruptions gracefully with reconnection logic
    - Send keep-alive messages during periods of silence
    - Close connections properly to free server resources
  </Accordion>
  
  <Accordion title="Audio Streaming" icon="waveform">
    - Send audio chunks at regular intervals (every 100-200ms)
    - Use appropriate chunk sizes (1024-2048 bytes recommended)
    - Maintain consistent sample rate throughout session
    - Handle audio buffer overflow/underflow conditions
  </Accordion>
  
  <Accordion title="Message Processing" icon="envelope">
    - Process messages in order using message_id and timestamp
    - Handle partial messages for streaming responses
    - Implement message acknowledgment for critical operations
    - Cache messages for replay or debugging functionality
  </Accordion>
</AccordionGroup>
